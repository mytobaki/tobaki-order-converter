export default async function handler(req, res) { if (req.method !== 'POST') { return res.status(405).json({ error: 'Method not allowed' }); } const { fileContent } = req.body; try { const response = await fetch('https://api.anthropic.com/v1/messages', { method: 'POST', headers: { 'Content-Type': 'application/json', 'anthropic-version': '2023-06-01' }, body: JSON.stringify({ model: 'claude-opus-4-1', max_tokens: 2000, messages: [{ role: 'user', content: `다음은 토박이마을 단체주문 데이터입니다. 이 데이터를 분석하고, JSON 배열로 정렬해주세요. 필수 열: No, 주문번호, 이름, 우편번호, 주소, 전화번호, 휴대폰번호, 주문내용, 수량, 배송메시지, 보내는사람 이름, 보내는사람 주소, 보내는사람 전화번호 데이터: ${fileContent.substring(0, 3000)} 응답은 다음 형식의 JSON만 반환하세요 (설명 없음): [ {"No": "1", "주문번호": "", "이름": "", "우편번호": "", "주소": "", "전화번호": "", "휴대폰번호": "", "주문내용": "", "수량": "", "배송메시지": "", "보내는사람 이름": "", "보내는사람 주소": "", "보내는사람 전화번호": ""} ]` }] }) }); if (!response.ok) { const error = await response.json(); return res.status(response.status).json({ error: error.error?.message || 'API error' }); } const data = await response.json(); const content = data.content[0].text; let jsonMatch = content.match(/\[[\s\S]*?\]/); if (!jsonMatch) { jsonMatch = content.match(/```json\n?([\s\S]*?)\n?```/); if (jsonMatch) { jsonMatch = [jsonMatch[1]]; } } if (!jsonMatch) { return res.status(400).json({ error: 'JSON을 찾을 수 없습니다' }); } const parsed = JSON.parse(jsonMatch[0]); const rows = Array.isArray(parsed) ? parsed : [parsed]; res.status(200).json({ success: true, data: rows }); } catch (error) { console.error('Error:', error); res.status(500).json({ error: error.message }); } }
